def is_automation_canvas_fingerprint(canvas_data):
    """
    Détection d'empreintes Canvas typiques de l'automation
    """
    if not canvas_data:
        return False
    
    # Signatures connues d'automation
    automation_signatures = [
        'da39a3ee5e6b4b0d3255bfef95601890afd80709',  # Canvas vide
        '2d2d2d2d2d2d2d2d2d2d2d2d2d2d2d2d2d2d2d2d',  # Pattern répétitif
        # Autres signatures connues...
    ]
    
    canvas_hash = canvas_data.get('hash', '')
    
    # Vérification des signatures connues
    if canvas_hash in automation_signatures:
        return True
    
    # Vérification de la complexité
    if len(set(canvas_hash)) < 8:  # Trop peu de caractères uniques
        return True
    
    # Vérification des dimensions suspectes
    dimensions = canvas_data.get('dimensions', '')
    if dimensions in ['1280x720', '1920x1080'] and canvas_data.get('is_default', False):
        return True
    
    return False

def has_automation_performance_signature(perf_timing):
    """
    Détection de signatures de performance typiques de l'automation
    """
    if not perf_timing:
        return False
    
    # Temps de chargement trop rapides/réguliers
    load_time = perf_timing.get('loadEventEnd', 0) - perf_timing.get('navigationStart', 0)
    
    # Automation souvent plus rapide et plus régulière
    if load_time < 100:  # Moins de 100ms
        return True
    
    # Vérification des temps de DNS/TCP trop parfaits
    dns_time = perf_timing.get('domainLookupEnd', 0) - perf_timing.get('domainLookupStart', 0)
    tcp_time = perf_timing.get('connectEnd', 0) - perf_timing.get('connectStart', 0)
    
    if dns_time == 0 and tcp_time == 0:  # Connexion réutilisée de manière suspecte
        return True

class AdaptiveAutomationChallenges:
    def __init__(self):
        self.challenge_types = [
            'mouse_tracking_challenge',
            'timing_based_challenge',
            'visual_puzzle_challenge',
            'behavioral_consistency_challenge',
            'multi_step_verification'
        ]
    
    def generate_anti_automation_challenge(self, automation_score, detected_tool):
        """
        Génération de défis spécialisés anti-automation
        """
        if detected_tool == 'puppeteer':
            return self.generate_puppeteer_specific_challenge(automation_score)
        elif detected_tool == 'playwright':
            return self.generate_playwright_specific_challenge(automation_score)
        else:
            return self.generate_generic_automation_challenge(automation_score)
    
    def generate_puppeteer_specific_challenge(self, score):
        """
        Défis spécialisés contre Puppeteer
        """
        challenges = []
        
        # Défi 1: Détection de webdriver property
        challenges.append({
            'type': 'webdriver_detection',
            'javascript': '''

                    return {detected: true, tool: 'puppeteer'};
                }
                return {detected: false};
            ''',
            'expected_response': {'detected': False}
        })
        
        # Défi 2: Test de permissions API
        challenges.append({
            'type': 'permissions_api_test',

                    .then(result => ({
                        state: result.state,
                        timestamp: Date.now()
                    }));
            ''',
            'validation': 'check_permission_timing'
        })
        
        # Défi 3: Mouse movement entropy
        challenges.append({
            'type': 'mouse_entropy_challenge',
            'instructions': 'Move mouse in natural pattern for 3 seconds',
            'duration': 3000,
            'validation': 'analyze_mouse_entropy'
        })
        
        return {
            'challenge_id': self.generate_challenge_id(),
            'challenges': challenges,
            'timeout': 30000,
            'required_success_rate': 0.8
        }
    
    def generate_playwright_specific_challenge(self, score):
        """
        Défis spécialisés contre Playwright
        """
        challenges = []
        
        # Défi 1: Détection de variables globales Playwright
        challenges.append({
            'type': 'playwright_globals_detection',
            'javascript': '''
                const playwrightIndicators = [
                    'window.__playwright',
                    'window.__pw_manual',
                    'window.playwright'
                ];
                
                for (let indicator of playwrightIndicators) {
                    if (eval(indicator)) {
                        return {detected: true, indicator: indicator};
                    }
                }
                return {detected: false};
            ''',
            'expected_response': {'detected': False}
        })
        
        # Défi 2: Test de viewport par défaut
        challenges.append({
            'type': 'viewport_test',
            'javascript': '''
                return {
                    screenWidth: screen.width,
                    screenHeight: screen.height,
                    windowWidth: window.innerWidth,
                    windowHeight: window.innerHeight,
                    isDefault: (screen.width === 1280 && screen.height === 720)
                };
            ''',
            'validation': 'check_viewport_naturalness'
        })
        
        # Défi 3: Test de CDP (Chrome DevTools Protocol)
        challenges.append({
            'type': 'cdp_detection',
            'javascript': '''
                return {
                    hasDebugger: !!(window.chrome && window.chrome.debugger),
                    hasRuntime: !!(window.chrome && window.chrome.runtime),
                    timestamp: Date.now()
                };
            ''',
            'validation': 'analyze_cdp_presence'
        })
        
        return {
            'challenge_id': self.generate_challenge_id(),
            'challenges': challenges,
            'timeout': 25000,
            'required_success_rate': 0.85
        }
    
    def validate_challenge_response(self, challenge_id, responses):
        """
        Validation sophistiquée des réponses aux défis
        """
        challenge = self.get_challenge(challenge_id)
        validation_results = []
        
        for i, response in enumerate(responses):
            challenge_def = challenge['challenges'][i]
            
            if challenge_def['type'] == 'mouse_entropy_challenge':
                result = self.validate_mouse_entropy(response)
            elif challenge_def['type'] == 'webdriver_detection':
                result = self.validate_webdriver_response(response, challenge_def)
            elif challenge_def['type'] == 'viewport_test':
                result = self.validate_viewport_response(response)
            else:
                result = self.validate_generic_response(response, challenge_def)
            
            validation_results.append(result)
        
        # Calcul du score global
        success_rate = sum(r['success'] for r in validation_results) / len(validation_results)
        
        return {
            'challenge_id': challenge_id,
            'success_rate': success_rate,
            'individual_results': validation_results,
            'passed': success_rate >= challenge['required_success_rate'],
            'automation_likelihood': 1 - success_rate
        }
    
    def validate_mouse_entropy(self, mouse_data):
        """
        Validation de l'entropie des mouvements de souris
        """
        if not mouse_data or len(mouse_data) < 10:
            return {'success': False, 'reason': 'insufficient_mouse_data'}
        
        # Calcul de l'entropie
        entropy = self.calculate_mouse_entropy(mouse_data)
        
        # Détection de patterns linéaires
        linear_ratio = self.detect_linear_mouse_patterns(mouse_data)
        
        # Analyse de la vitesse
        velocity_variance = self.calculate_velocity_variance(mouse_data)
        
        # Score composite
        human_score = (entropy * 0.4) + ((1 - linear_ratio) * 0.3) + (velocity_variance * 0.3)
        
        return {
            'success': human_score > 0.6,
            'human_score': human_score,
            'entropy': entropy,
            'linear_ratio': linear_ratio,
            'velocity_variance': velocity_variance
        }
