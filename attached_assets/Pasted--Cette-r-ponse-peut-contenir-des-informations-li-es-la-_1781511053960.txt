def detect_http1_downgrade_attack(request):

    Détection du downgrade HTTP/1.1 forcé

    suspicious_indicators = []

import json
import boto3
from datetime import datetime, timedelta
import hashlib

dynamodb = boto3.resource('dynamodb')
session_table = dynamodb.Table('agotels-session-tracking')

def lambda_handler(event, context):
    """
    Détecteur d'attaques split-traffic sophistiqué
    """
    request_data = extract_request_data(event)
    
    # Analyse de corrélation de session
    session_correlation = analyze_session_correlation(request_data)
    
    # Détection de saut d'IP rapide
    ip_jump_analysis = detect_rapid_ip_changes(request_data)
    
    # Analyse de cohérence géographique
    geo_consistency = analyze_geo_consistency(request_data)
    
    # Score composite de split-traffic
    split_traffic_score = calculate_split_traffic_score(

    )
    
    if split_traffic_score > SPLIT_TRAFFIC_THRESHOLD:
        return block_split_traffic_attack(request_data, split_traffic_score)
    
    return allow_request(request_data)

def analyze_session_correlation(request_data):
    """
    Analyse de corrélation de session avancée
    """
    session_token = request_data.get('session_token')
    current_ip = request_data.get('source_ip')
    timestamp = datetime.utcnow()
    
    # Récupération de l'historique de session
    session_history = get_session_history(session_token)
    
    correlation_signals = {
        'ip_changes': count_ip_changes(session_history),
        'time_between_ips': calculate_ip_change_timing(session_history),
        'geographic_jumps': detect_geographic_jumps(session_history),
        'device_fingerprint_consistency': check_device_consistency(session_history),
        'behavioral_continuity': analyze_behavioral_continuity(session_history)
    }
    
    # Détection de patterns suspects
    suspicious_patterns = []
    
    # Pattern 1: Changement d'IP trop rapide
    if correlation_signals['time_between_ips'] < timedelta(seconds=30):
        suspicious_patterns.append({
            'type': 'rapid_ip_change',
            'severity': 'high',
            'evidence': correlation_signals['time_between_ips']
        })
    
    # Pattern 2: Saut géographique impossible
    if correlation_signals['geographic_jumps']:
        for jump in correlation_signals['geographic_jumps']:
            if jump['distance'] > 1000 and jump['time'] < timedelta(minutes=5):
                suspicious_patterns.append({
                    'type': 'impossible_geographic_jump',
                    'severity': 'critical',
                    'evidence': jump
                })
    
    # Pattern 3: Incohérence d'empreinte device
    if not correlation_signals['device_fingerprint_consistency']:
        suspicious_patterns.append({
            'type': 'device_fingerprint_mismatch',
            'severity': 'medium',
            'evidence': 'fingerprint_changed_with_ip'
        })
    
    return {
        'signals': correlation_signals,
        'suspicious_patterns': suspicious_patterns,
        'risk_score': calculate_correlation_risk_score(suspicious_patterns)
    }

def detect_rapid_ip_changes(request_data):
    """
    Détection spécialisée des changements d'IP rapides
    """
    session_token = request_data.get('session_token')
    current_ip = request_data.get('source_ip')
    
    # Fenêtre d'analyse glissante
    time_window = timedelta(minutes=10)
    recent_ips = get_recent_session_ips(session_token, time_window)
    
    rapid_change_indicators = {
        'ip_count_in_window': len(recent_ips),
        'average_time_per_ip': calculate_average_ip_duration(recent_ips),
        'ip_type_changes': detect_ip_type_changes(recent_ips),
        'asn_changes': detect_asn_changes(recent_ips),
        'residential_to_datacenter': detect_ip_type_escalation(recent_ips)
    }
    
    # Calcul du score de risque
    risk_factors = []
    
    if rapid_change_indicators['ip_count_in_window'] > 3:
        risk_factors.append(('multiple_ips', 40))
    
    if rapid_change_indicators['average_time_per_ip'] < timedelta(seconds=60):
        risk_factors.append(('very_rapid_changes', 50))
    
    if rapid_change_indicators['residential_to_datacenter']:
        risk_factors.append(('ip_type_escalation', 60))
    
    return {
        'indicators': rapid_change_indicators,
        'risk_factors': risk_factors,
        'risk_score': sum(score for _, score in risk_factors)
    }

def analyze_geo_consistency(request_data):
    """
    Analyse de cohérence géographique sophistiquée
    """
    session_token = request_data.get('session_token')
    current_ip = request_data.get('source_ip')
    
    # Géolocalisation précise
    current_geo = get_precise_geolocation(current_ip)
    session_geo_history = get_session_geo_history(session_token)
    
    geo_analysis = {
        'distance_from_last': calculate_distance(
            current_geo, session_geo_history[-1] if session_geo_history else None
        ),
        'travel_time_feasibility': check_travel_feasibility(
            current_geo, session_geo_history
        ),
        'timezone_consistency': check_timezone_consistency(
            current_geo, request_data.get('timestamp')
        ),
        'isp_consistency': check_isp_consistency(current_ip, session_geo_history)
    }
    
    # Détection d'anomalies géographiques
    geo_anomalies = []
    
    if geo_analysis['distance_from_last'] > 500:  # km
        travel_time = geo_analysis['travel_time_feasibility']
        if not travel_time['feasible']:
            geo_anomalies.append({
                'type': 'impossible_travel',
                'distance': geo_analysis['distance_from_last'],
                'time_required': travel_time['minimum_time'],
                'actual_time': travel_time['actual_time']
            })
    
    return {
        'analysis': geo_analysis,
        'anomalies': geo_anomalies,
        'risk_score': calculate_geo_risk_score(geo_anomalies)
    }
