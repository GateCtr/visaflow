def extract_bigip_server_cookie(headers):
    """
    Extraction et analyse du cookie BIGipServer
    """
    bigip_cookie = headers.get('Cookie', '').split('BIGipServer')
    if len(bigip_cookie) > 1:
        cookie_value = bigip_cookie[1].split(';')[0].strip('=')
        
        # Décodage de la valeur F5
        decoded_info = decode_f5_cookie_value(cookie_value)
        
        return {
            'raw_value': cookie_value,
            'decoded_ip': decoded_info.get('ip'),
            'decoded_port': decoded_info.get('port'),
            'encoding_method': decoded_info.get('method'),
            'is_valid_encoding': validate_f5_encoding(cookie_value)
        }
    
    return None
