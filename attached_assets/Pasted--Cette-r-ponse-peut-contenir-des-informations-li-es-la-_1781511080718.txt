            try {
                const audioContext = new (window.AudioContext || window.webkitAudioContext)();
                const oscillator = audioContext.createOscillator();
                const analyser = audioContext.createAnalyser();
                const gainNode = audioContext.createGain();

        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            const char = str.charCodeAt(i);
            hash = ((hash << 5) - hash) + char;
            hash = hash & hash; // Convert to 32bit integer
        }

    def generate_dynamic_challenge(self, session_id, difficulty_level):
        """

        """
        challenge_types = [
            'proof_of_work',
            'memory_hard_function',
            'time_lock_puzzle',
            'interactive_proof'
        ]
        
        challenge_type = self.select_challenge_type(difficulty_level)
        
        if challenge_type == 'proof_of_work':
            return self.generate_proof_of_work_challenge(session_id, difficulty_level)
        elif challenge_type == 'memory_hard_function':
            return self.generate_memory_hard_challenge(session_id, difficulty_level)
        elif challenge_type == 'time_lock_puzzle':
            return self.generate_time_lock_challenge(session_id, difficulty_level)
        else:
            return self.generate_interactive_proof_challenge(session_id, difficulty_level)
    
    def generate_proof_of_work_challenge(self, session_id, difficulty):
        """
        Défi Proof-of-Work adaptatif
        """
        nonce = secrets.token_hex(16)
        target_prefix = '0' * difficulty
        
        challenge = {
            'type': 'proof_of_work',
            'session_id': session_id,
            'nonce': nonce,
            'target_prefix': target_prefix,
            'algorithm': 'sha256',
            'timestamp': time.time(),
            'expected_time': self.calculate_expected_solve_time(difficulty)
        }
        
        self.active_challenges[session_id] = challenge
        
        return {
            'challenge_id': session_id,
            'nonce': nonce,
            'target_prefix': target_prefix,
            'instructions': f'Find a value that when concatenated with {nonce} and hashed with SHA256 starts with {target_prefix}'
        }
    
    def verify_challenge_response(self, session_id, response):
        """
        Vérification sophistiquée des réponses
        """
        if session_id not in self.active_challenges:
            return {'valid': False, 'reason': 'challenge_not_found'}
        
        challenge = self.active_challenges[session_id]
        
        # Vérification temporelle
        solve_time = time.time() - challenge['timestamp']
        expected_time = challenge['expected_time']
        
        # Détection de résolution trop rapide (bot)
        if solve_time < expected_time * 0.1:
            return {
                'valid': False, 
                'reason': 'solved_too_quickly',
                'solve_time': solve_time,
                'expected_minimum': expected_time * 0.1
            }
        
        # Détection de résolution trop lente (timeout)
        if solve_time > expected_time * 10:
            return {
                'valid': False,
                'reason': 'timeout',
                'solve_time': solve_time,
                'maximum_allowed': expected_time * 10
            }
        
        # Vérification cryptographique
        if challenge['type'] == 'proof_of_work':
            return self.verify_proof_of_work(challenge, response, solve_time)
        
        return {'valid': False, 'reason': 'unknown_challenge_type'}
    
    def verify_proof_of_work(self, challenge, response, solve_time):
        """
        Vérification Proof-of-Work avec analyse comportementale
        """
        combined = challenge['nonce'] + response
        hash_result = hashlib.sha256(combined.encode()).hexdigest()
        
        is_valid = hash_result.startswith(challenge['target_prefix'])
        
        if is_valid:
            # Analyse comportementale du temps de résolution
            behavioral_analysis = self.analyze_solve_time_behavior(
                challenge['expected_time'], 
                solve_time,
                challenge['session_id']
            )
            
            return {
                'valid': True,
                'solve_time': solve_time,
                'hash_result': hash_result,
                'behavioral_score': behavioral_analysis['human_likelihood'],
                'automation_indicators': behavioral_analysis['automation_signals']
            }
        
        return {'valid': False, 'reason': 'incorrect_solution'}

def analyze_mobile_specific_signals(request_data):
    """
    Analyse spécialisée pour les appareils mobiles
    """
    mobile_signals = {
        'device_orientation': extract_device_orientation(request_data),
        'touch_events': analyze_touch_events(request_data),
        'accelerometer_data': extract_accelerometer_data(request_data),
        'battery_api': analyze_battery_api_usage(request_data),
        'network_type': detect_network_type(request_data),
        'app_vs_browser': detect_app_vs_browser_context(request_data)
    }
    
    # Détection d'émulation mobile
    emulation_indicators = []
    
    # Indicateur 1: Orientation sans événements tactiles
    if mobile_signals['device_orientation'] and not mobile_signals['touch_events']:
        emulation_indicators.append({
            'type': 'orientation_without_touch',
            'severity': 'medium',
            'description': 'Device orientation detected without corresponding touch events'
        })
    
    # Indicateur 2: User-Agent mobile avec comportement desktop
    if is_mobile_user_agent(request_data) and has_desktop_behavior(mobile_signals):
        emulation_indicators.append({
            'type': 'mobile_ua_desktop_behavior',
            'severity': 'high',
            'description': 'Mobile User-Agent with desktop-like behavior patterns'
        })

    if has_inconsistent_mobile_apis(mobile_signals):
        emulation_indicators.append({
            'type': 'inconsistent_mobile_apis',
            'severity': 'high',
            'description': 'Inconsistent mobile API availability or behavior'
        })
    
    return {
        'mobile_signals': mobile_signals,
        'emulation_indicators': emulation_indicators,
        'mobile_authenticity_score': calculate_mobile_authenticity_score(
            mobile_signals, emulation_indicators
