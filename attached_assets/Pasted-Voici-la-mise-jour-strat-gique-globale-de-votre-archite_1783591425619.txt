1. Mise à jour du script WhatsApp (Bouton Pré-configuré)
Le bouton utilise la structure d'URL officielle. Le message est formaté pour que le client qualifie son besoin dès le premier clic, vous évitant de perdre du temps en discussions inutiles.
// Composant Bouton WhatsApp pour votre Landing Page React
import React from 'react';
import { MessageSquare } from 'lucide-react';

export default function WhatsAppCTA() {
  // Numéro officiel Joventy extrait du site
  const whatsappNumber = "243899450845"; 
  
  const whatsappMessage = encodeURIComponent(
    "Bonjour Joventy, je viens du site. Je prépare un dossier de visa et je souhaite commander un Audit & Diagnostic pour sécuriser ma demande."
  );
  const whatsappUrl = `https://wa.me/${whatsappNumber}?text=${whatsappMessage}`;

  return (
    <a 
      href={whatsappUrl}
      target="_blank"
      rel="noopener noreferrer"
      className="inline-flex items-center justify-center gap-3 bg-emerald-600 hover:bg-emerald-500 text-white font-extrabold px-8 py-4 rounded-xl shadow-xl transition-all transform hover:-translate-y-0.5 w-full sm:w-auto text-base tracking-wide"
    >
      <MessageSquare className="w-5 h-5 fill-current" />
      Faire analyser mon dossier sur WhatsApp
    </a>
  );
}
